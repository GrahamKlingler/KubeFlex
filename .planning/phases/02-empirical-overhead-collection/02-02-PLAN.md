4. Helper functions:

`kubectl(*args, capture=True, timeout=60)` — runs kubectl command via `subprocess.run(["kubectl"] + list(args), capture_output=capture, text=True, timeout=timeout)`. Returns `(returncode, stdout, stderr)`. Wraps in try/except for TimeoutExpired.

`get_migrate_pod_name(namespace)` — runs `kubectl get pod -n {namespace} -l app=python-migrate -o jsonpath='{.items[0].metadata.name}'` to dynamically find the migration service pod name (per RESEARCH.md Open Question 2).

`wait_for_pod_ready(name, namespace, timeout=120)` — runs `kubectl wait --for=condition=Ready pod/{name} -n {namespace} --timeout={timeout}s`. Returns bool.

`cleanup_workload_pods(namespace)` — deletes all pods matching workload basenames (`sysbench-cpu-pod*`, `sysbench-memory-pod*`, `memcount-pod*`) in namespace. Uses `kubectl delete pod -n {namespace} -l name={label} --ignore-not-found`. Also cleans up checkpoint directories on all three worker nodes: `kubectl exec -n monitor migrator-kind-worker -- rm -rf /tmp/checkpoints/migration_checkpoint` (per RESEARCH.md Pitfall 2).

`deploy_workload(workload_key, source_region)` — reads the YAML template from `WORKLOAD_YAMLS[workload_key]` (path relative to script directory), sed-replaces `kubernetes.io/hostname: kind-worker` with the correct node for `source_region`, applies via `kubectl apply -f -` with stdin. Returns the pod name.

`extract_timing_from_logs(pod_name, namespace, since_time)` — runs `kubectl logs {pod_name} -n {namespace} --since-time={since_time}`. Parses each line: find first `{` character, attempt `json.loads(line[idx:])`, check for `event` key in `("checkpoint_complete", "transfer_complete", "restore_complete")`. Returns dict mapping event name to duration_ms. If more than 3 events found, log a warning (Pitfall 1).

`perform_single_migration(migration_service_url, pod_name, source_node, target_node, target_region, namespace)` — captures ISO timestamp just before the call. POSTs to `{migration_service_url}/live-migrate` with JSON body `{"namespace": namespace, "pod": pod_name, "source_node": source_node, "target_node": target_node, "target_region": target_region, "delete_original": True}` and `timeout=300`. Returns `(success: bool, response_json: dict)`.

`write_csv_row(csv_path, row_dict)` — opens file in append mode (`"a"`), creates `csv.DictWriter(f, fieldnames=CSV_FIELDS)`, writes header if `f.tell() == 0`, writes row. This ensures partial runs are recoverable (per RESEARCH.md anti-pattern warning).

`write_metadata_json(json_path, run_id, config, rows)` — writes full run metadata: `{"run_id": run_id, "config": config, "total_migrations": len(rows), "successful": sum(r["success"] for r in rows), "migrations": rows}`.

5. Main benchmark loop `run_benchmark(args)`:
```
run_id = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
out_dir = Path(args.out_dir) if args.out_dir else script_dir.parent.parent.parent / "data" / "overhead-benchmark" / f"run_{run_id}"
out_dir.mkdir(parents=True, exist_ok=True)
csv_path = out_dir / "overhead_raw.csv"
json_path = out_dir / "overhead_metadata.json"

migrate_pod = get_migrate_pod_name(args.namespace)
migration_num = 0
all_rows = []

for workload in WORKLOADS:
    for (src_region, tgt_region) in REGION_PAIRS:
        for rep in range(1, args.repetitions + 1):
            migration_num += 1
            logger.info("=" * 80)
            logger.info(f"[BENCHMARK] Migration {migration_num}: {workload} {src_region}->{tgt_region} rep {rep}")
            
            # 1. Cleanup previous pods and checkpoint dirs
            cleanup_workload_pods(args.namespace)
            
            # 2. Deploy workload pod to source node
            pod_name = deploy_workload(workload, src_region)
            
            # 3. Wait for pod Ready
            if not wait_for_pod_ready(pod_name, args.namespace):
                logger.error(f"[BENCHMARK] Pod {pod_name} not ready, skipping")
                # write failure row
                row = {field: "" for field in CSV_FIELDS}
                row.update({"run_id": run_id, "migration_num": migration_num, ...failure fields...})
                write_csv_row(csv_path, row)
                all_rows.append(row)
                continue
            
            # 4. Allow workload to start (sleep 5s for memcount to allocate memory)
            time.sleep(5)
            
            # 5. Capture timestamp for log filtering
            since_time = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
            
            # 6. POST /live-migrate
            source_node = REGION_TO_NODE[src_region]
            target_node = REGION_TO_NODE[tgt_region]
            success, resp = perform_single_migration(
                args.migration_service_url, pod_name, source_node, target_node, tgt_region, args.namespace
            )
            
            # 7. Extract timing from logs
            timings = {}
            if success:
                time.sleep(2)  # brief pause for log flush
                timings = extract_timing_from_logs(migrate_pod, args.namespace, since_time)
            
            # 8. Build and write CSV row
            row = {
                "run_id": run_id,
                "migration_num": migration_num,
                "workload": workload,
                "source_region": src_region,
                "target_region": tgt_region,
                "repetition": rep,
                "checkpoint_ms": timings.get("checkpoint_complete", ""),
                "transfer_ms": timings.get("transfer_complete", ""),
                "restore_ms": timings.get("restore_complete", ""),
                "total_ms": sum(timings.values()) if len(timings) == 3 else "",
                "success": str(success).lower(),
                "timestamp_utc": datetime.now(timezone.utc).isoformat(),
            }
            write_csv_row(csv_path, row)
            all_rows.append(row)
            
            logger.info(f"[BENCHMARK] Result: success={success} timings={timings}")
```

```python
#!/usr/bin/env python3
"""Unit tests for overhead benchmark CSV schema validation."""

import csv
import sys
from pathlib import Path


REQUIRED_FIELDS = {
    "run_id", "migration_num", "workload",
    "source_region", "target_region", "repetition",
    "checkpoint_ms", "transfer_ms", "restore_ms", "total_ms",
    "success", "timestamp_utc",
}

VALID_WORKLOADS = {"sysbench_cpu", "sysbench_memory", "memcount"}
VALID_REGIONS = {"NE", "TEN", "CENT"}


def test_csv_has_required_fields(csv_path):
    """Verify CSV header contains all required fields."""
    with open(csv_path) as f:
        reader = csv.DictReader(f)
        fields = set(reader.fieldnames)
    missing = REQUIRED_FIELDS - fields
    assert not missing, f"Missing fields: {missing}"
    print(f"  Header fields: {sorted(fields)}")


def test_total_ms_equals_sum_of_phases(csv_path):
    """For successful rows, total_ms should equal checkpoint_ms + transfer_ms + restore_ms."""
    with open(csv_path) as f:
        reader = csv.DictReader(f)
        for i, row in enumerate(reader):
            if row["success"] != "true":
                continue
            ckpt = float(row["checkpoint_ms"])
            xfer = float(row["transfer_ms"])
            rest = float(row["restore_ms"])
            total = float(row["total_ms"])
            expected = ckpt + xfer + rest
            assert abs(total - expected) < 0.1, (
                f"Row {i}: total_ms={total} != sum={expected}"
            )


def test_valid_workload_names(csv_path):
    """Verify workload column contains only known workload names."""
    with open(csv_path) as f:
        reader = csv.DictReader(f)
        for i, row in enumerate(reader):
            assert row["workload"] in VALID_WORKLOADS, (
                f"Row {i}: unknown workload '{row['workload']}'"
            )


def test_valid_regions(csv_path):
    """Verify source_region and target_region contain valid region names."""
    with open(csv_path) as f:
        reader = csv.DictReader(f)
        for i, row in enumerate(reader):
            assert row["source_region"] in VALID_REGIONS, (
                f"Row {i}: invalid source_region '{row['source_region']}'"
            )
            assert row["target_region"] in VALID_REGIONS, (
                f"Row {i}: invalid target_region '{row['target_region']}'"
            )
            assert row["source_region"] != row["target_region"], (
                f"Row {i}: source and target region are the same"
            )
```
